title: Anomalous Bulk Customer DB Query Followed by USB Mass Storage Connection
id: 8f3a9c12-4b7e-4d91-a2c5-7e1f9b8d3e04
status: experimental
description: >
  Detects potential insider data exfiltration where a non-privileged business account
  performs a bulk SELECT query on the customer database followed within a short timeframe
  by a removable USB mass storage connection on the endpoint.
references:
  - https://attack.mitre.org/techniques/T1078/
  - https://attack.mitre.org/techniques/T1052/001/
author: Security Operations Portfolio (Sanitized Case Study)
date: 2026-04-01
tags:
  - attack.initial_access
  - attack.t1078
  - attack.exfiltration
  - attack.t1052.001
logsource:
  category: correlation
  product: siem
detection:
  selection_db_audit:
    event.category: 'database'
    db.operation: 'SELECT'
    db.table: 'customers'
    user.role:
      - 'sales'
      - 'support'
      - 'intern'
    db.rows_returned|gte: 500
  selection_endpoint_usb:
    event.category: 'driver'
    device.class: 'USB_MASS_STORAGE'
    event.action: 'device_connected'
  correlation:
    type: temporal
    by:
      - user.name
      - host.name
    timespan: 15m
    condition: selection_db_audit and selection_endpoint_usb
falsepositives:
  - Authorized periodic reporting exports (should be performed via CRM reporting module, not direct SQL)
  - IT administrators performing approved backups (excluded via user.role filter)
level: high
